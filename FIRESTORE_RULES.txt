rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    function isSignedIn() {
      return request.auth != null;
    }
    function getUserRole(uid) {
      return get(/databases/$(database)/documents/users/$(uid)).data.role;
    }
    function isAdmin() {
      return isSignedIn() && getUserRole(request.auth.uid) == 'admin';
    }

    // Invitaciones para registro de profesionales
    match /invitations/{token} {
      // Solo administradores pueden crear invitaciones
      allow create: if isAdmin();

      // Lectura permitida si tienes el token (secreto compartido por el admin)
      allow get, list: if true;

      // Actualizar: admin o el propio invitado puede marcarla como usada
      allow update: if (
        isAdmin()
      ) || (
        isSignedIn()
        && resource.data.used == false
        && request.resource.data.used == true
        && request.resource.data.usedBy == request.auth.uid
        && request.resource.data.email == resource.data.email
        && request.resource.data.name == resource.data.name
        && request.resource.data.role == resource.data.role
        && request.resource.data.specialty == resource.data.specialty
        && request.resource.data.matricula == resource.data.matricula
        && request.resource.data.phone == resource.data.phone
        && request.resource.data.workDays == resource.data.workDays
        && request.resource.data.workHours == resource.data.workHours
      );

      // Solo admin puede eliminar
      allow delete: if isAdmin();
    }

    // Colección de usuarios (perfiles)
    match /users/{userId} {
      allow read: if isSignedIn() && (request.auth.uid == userId || isAdmin());

      // Creación de perfil de usuario (registro)
      allow create: if (
        // El usuario se está registrando con su propio UID
        isSignedIn() && request.auth.uid == userId
        && (
          // Registro público de pacientes
          request.resource.data.role == 'patient'
          // O registro mediante invitación válida (ej: médico)
          || (
            request.resource.data.inviteToken != null
            && exists(/databases/$(database)/documents/invitations/$(request.resource.data.inviteToken))
            && get(/databases/$(database)/documents/invitations/$(request.resource.data.inviteToken)).data.email == request.auth.token.email
            && get(/databases/$(database)/documents/invitations/$(request.resource.data.inviteToken)).data.used == false
            && request.resource.data.role == get(/databases/$(database)/documents/invitations/$(request.resource.data.inviteToken)).data.role
          )
        )
      );

      // Actualización del propio perfil: no se permite escalar el role desde el cliente
      allow update: if (
        isSignedIn() && request.auth.uid == userId && request.resource.data.role == resource.data.role
      ) || isAdmin();

      // Solo administradores pueden eliminar usuarios
      allow delete: if isAdmin();
    }

    // Ejemplo simple para appointments
    match /appointments/{doc} {
      allow read: if isSignedIn();
      allow create: if isSignedIn();
      allow update, delete: if isAdmin() || resource.data.patientId == request.auth.uid || resource.data.doctorId == request.auth.uid;
    }

    // Reglas por defecto: denegar todo lo demás
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
